cmake_minimum_required(VERSION 3.16)
project(PrimeNumberTutor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 COMPONENTS Widgets QUIET)
if(Qt6_FOUND)
  set(QT_LIBS Qt6::Widgets)
else()
  find_package(Qt5 COMPONENTS Widgets REQUIRED)
  set(QT_LIBS Qt5::Widgets)
endif()
add_executable(prime-number-tutor src/main.cpp src/mainwindow.cpp src/mainwindow.h)
target_link_libraries(prime-number-tutor PRIVATE ${QT_LIBS})
install(TARGETS prime-number-tutor RUNTIME DESTINATION bin)
